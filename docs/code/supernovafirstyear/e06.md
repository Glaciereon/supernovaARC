float getDeltaHeight_seperated(float height, float velocity, float dragCoeficciant) {
  float delta_height = 0;
  float dt = 0.25;
 
  // Simulate the height change while velocity is greater than 1
  while (velocity > 1) {
    float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
    float acceleration = force / mass;
    velocity -= acceleration * dt;
    delta_height += velocity * dt;
  }


  return delta_height;
}


float getDeltaHeight_together(float height, float velocity, float dragCoeficciant, float coastTime) {
  float time = 0;
  float dt = 0.05;
  float delta_height = 0;
 
  // Simulate the height change during coasting phase
  while (time < coastTime) {
    float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
    float acceleration = force / mass;
    velocity -= acceleration * dt;
    delta_height += velocity * dt;
    time += dt;
  }


  return delta_height;
}


double pTime() {
  return ((double)(millis())) / (double)(1000); // Get time in seconds
}


float averageLastFive(float data[], int i) {
  // Ensure that the index is large enough to compute the average of the last 5 values
  if (i < 4) {
    Serial.println("Index is too small to compute average of last 5 values.");
    return -1;  // Return -1 or any error value to indicate invalid input
  }


  float sum = 0;
 
  // Sum the last 5 values starting from index (i-4) to i
  for (int j = i - 4; j <= i; j++) {
    sum += data[j];
  }


  // Return the average of the last 5 values
  return sum / 5.0;
}
