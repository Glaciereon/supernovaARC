# e09: Apollo data extraction

```cpp
//e09: companion to v2, prints the saved heights and times back out of the eeprom

#include <EEPROM.h>


void setup() {
 Serial.begin(115200);
}


void loop() {  
delay(5000);
Serial.println("EEPROM DATA TRANSMISSION BEGIN: ");
for (int i = 0; i < EEPROM.length()/2; i++) {
   Serial.println("The rocket was at the height of " + (String)EEPROM.read(i+512) + "meters at " + (String)((double)EEPROM.read(i)/10) + " seconds");
 }
Serial.println("EEPROM DATA DONE");
}
```
