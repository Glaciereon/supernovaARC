```cpp
//e01: scratch file for figuring out how lists/arrays work, not rocket code

#include <list>

int main() {
    cout << "Hello World!" << endl;
    
    int dataList[3];
    dataList[2] = 752;
    dataList[0] = 432;
    dataList[1] = 23;
    int lstLength = sizeof(dataList)/sizeof(dataList[0]);

    for (int i = 0; i<lstLength; i++) {
        cout << "index " << i << " : " << dataList[i] << endl;
    }



    return 0;
}
```
