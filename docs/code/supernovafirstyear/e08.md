# e08: Apollo v2 data reset

```cpp
//e08: companion to v2, fills the whole eeprom with 254 so old data is cleared before a flight

#include <EEPROM.h>


void setup() {
Serial.begin(115200);
 for (int i = 0; i < EEPROM.length(); i++) {
   EEPROM.write(i, 254);
 }
 Serial.println("EEPROM DATA CLEARING BEGIN");
for (int i = 0; i < EEPROM.length(); i++) {
   Serial.println(EEPROM.read(i));
 }
Serial.println("EEPROM DATA CLEARED");
}


void loop() {
 // Clear eeprom
delay(10000);
}
```
