    //e02: pulls the apogee prediction out into its own sketch (two recursive functions) so the calc time can be measured

const float airDensity = 1.297;         // density in kg/m^3
const float referenceArea = .038;       // m/2
const float mass = .619;                // mass of rocket in kg 😊
const float dragCoeficciant = 1.31;     // in a number


void setup() {
 // put your setup code here, to run once:
 Serial.begin(9600);
}


void loop() {
 float height = 200;
 float time_0 = pTime();
 float velocity = 8.7374;
 for (int i = 0; i<1; i++) {
 height += getDeltaHeight_together(height, velocity, 0.073, 0);
 height += getDeltaHeight_seperated(height, velocity, 1.32);
}


 Serial.println((String)height);
Serial.println(pTime()-time_0);
delay(1000);
}


float getDeltaHeight_seperated(float height, float velocity, float dragCoeficciant) {
 if (velocity < 1) {
   return 0.0;
 }


 float force = .5 * dragCoeficciant * 0.038 /*referenceArea*/ * airDensity * velocity * velocity;
 float acceleration = force / 0.619 /*mass*/;
 velocity -= acceleration * .25;
 float delta_height = 0;
 delta_height = delta_height + velocity * .25 + getDeltaHeight_seperated(height, velocity, 1.32);
 return delta_height;
}


float getDeltaHeight_together(float height, float velocity, float dragCoeficciant, float time) {
 if (time >= .71) {
   return 0.0;
 }


 float force = .5 * dragCoeficciant * 0.038 /*referenceArea*/ * airDensity * velocity * velocity;
 float acceleration = force / 0.619 /*mass*/;
 velocity -= acceleration * .05;
 time += .05;
 float delta_height = 0;
 delta_height = delta_height + velocity * .05 + getDeltaHeight_together(height, velocity, .073, time);
 return delta_height;
}


 double pTime() {
return ((double)(millis())) / (double)(1000);
}
