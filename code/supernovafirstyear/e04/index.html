//object creation
Adafruit_BMP3XX bmp;
Servo myServo;


//constants
#define SEALEVELPRESSURE_HPA (1013.25)
const int servoButton = 12;                                      // button that rotates the servo 90 degrees

const float emergencyTime = 4.9;                                 // Seconds before automatic split
const int hSeparation = 752;                                     // Separation height in feet

byte servoPos = 90;
int listIndex = 0;

void setup() {
 //pin/serial attachment

/*
checks if the button is down for .12 seconds
returns true if it is
*/
bool buttonPressed() {
 if (digitalRead(servoButton) != HIGH) {
   delay(25);
   if (digitalRead(servoButton) != HIGH) {
     delay(25);  // this is for accidental presses or accidental contact
     if (digitalRead(servoButton) != HIGH) {
       delay(100);
       if (digitalRead(servoButton) != HIGH) {
         Serial.println("Button pressed");
         delay(20);
         return true;
       }
     }
   }
 }
 return false;
}


/*
whis is what happens from the point that the rocket goes above 10 meters
*/
void timeToFly() {
 time_0 = pTime();                                        //time at launch
 height_0 = bmp.readAltitude(SEALEVELPRESSURE_HPA) + 10;  //height at the start of data collection
 do {
   height = getHeight(1);
   velocity = getVelocity(height);
   if (height > apogee) {
     apogee = height;
   }
   if (listIndex < 1024) {
     record(height, (pTime() - time_0));
   }
   tApogee = height;
   tApogee += getDeltaHeight_together(height, velocity, .073, 0);
   tApogee += getDeltaHeight_seperated(height, velocity, 1.32);


 } while ((tApogee >= (240 - getDeltaHeight_together(height, velocity, 0.073, 0.71- 0.045))) && ((pTime() - time_0) < emergencyTime) && (height < 200));
 Serial.print("done.");
 myServo.write(90);
 while (((pTime() - time_0) < dataTime) &&(listIndex < 1024)) {
     record(getHeight(2), (pTime() - time_0));
   }


 Serial.println("EEPROM DATA BEGIN");
 for (int i = 0; i < EEPROM.length(); i++) {
   Serial.println(EEPROM.read(i));
 }
 Serial.println("EEPROM DATA DONE");


 Serial.println("done with flight, moving on to apogee loop");


 while (true) {
   Serial.println("this is the apogee in meters: " + (String)apogee);
   delay(1000);
 }
}

/*
records the data
makes important assumption that the change in height per interval doesnt exceed 100!!
*/
void record(float height, double time) {
 String cplusplussucks = "Time from liftoff in seconds is: " + (String)(pTime() - time_0) + " height comparative to liftoff is: " + (String)height + "\n";
 Serial.print(cplusplussucks);
 if (listIndex == 0) {
   EEPROM.write(0, (int)time);
   EEPROM.write(512, (int)height);
 } else {
   EEPROM.write(listIndex, (double)(10 * time));  //records the dime in deciseconds
   EEPROM.write(listIndex + 512, (int)height);    //records the dime in deciseconds
 }
 listIndex++;
}


/*
returns the average of the past sampleLength readings
*/
float getHeight(int sampleLength) {
 float average = 0;
 for (int i = 0; i < sampleLength; i++) {
   average += bmp.readAltitude(SEALEVELPRESSURE_HPA) - height_0;
 }
 return average / sampleLength;
}




float getDeltaHeight_seperated(float height, float velocity, float dragCoeficciant) {
 if (velocity < .5) {
   return 0.0;
 }


 float force = .5 * dragCoeficciant * 0.038 /*referenceArea*/ * airDensity * velocity * velocity;
 float acceleration = force / 0.619 /*mass*/;
 velocity -= acceleration * .25;
 float delta_height = 0;
 delta_height = delta_height + velocity * .25 + getDeltaHeight_seperated(height, velocity, 1.32);
 return delta_height;
}


float getDeltaHeight_together(float height, float velocity, float dragCoeficciant, float time) {
 if (time >= .71) {
   return 0.0;
 }


 float force = .5 * dragCoeficciant * 0.038 /*referenceArea*/ * airDensity * velocity * velocity;
 float acceleration = force / 0.619 /*mass*/;
 velocity -= acceleration * .05;
 time += .05;
 float delta_height = 0;
 delta_height = delta_height + velocity * .05 + getDeltaHeight_together(height, velocity, .073, time);
 return delta_height;
}
