//libraries
#include <EEPROM.h>

#include <Adafruit_Sensor.h>
#include "Adafruit_BMP3XX.h"
#include <Servo.h>


//object creation
Adafruit_BMP3XX bmp;
Servo myServo;


//constants
#define SEALEVELPRESSURE_HPA (1013.25)
const int servoButton = 12;                                      // button that rotates the servo 90 degrees
const int servoPin = 4;                                          // Pin for servo motor
const float emergencyTime = 6;                                 // Seconds before automatic split
const float timeFromStartOfSeperationToUnfoldedParachute = .71;  //time From Start Of Seperation ToUnfolded Parachute in seconds
const int dataTime = 8;
const float airDensity = 1.297;    // density in kg/m^3
const float referenceArea = .038;  // m/2
const float mass = .645;           // mass of rocket in kg 😊
byte servoPos = 90;
int listIndex = 0;
float height = 0;
float height_0 = 0;
float previous_height = 0.0;
float time_0 = 0;
float apogee = 0;
float tApogee = 0;
float average = 0;
float velocity = 0;  // velocity in m/s
float height_list[5] = {0,0,0,0,0};


void setup() {
 //pin/serial attachment
 pinMode(13, OUTPUT);  //this is the onboard LED
 digitalWrite(13, LOW);
 pinMode(servoButton, INPUT);
 myServo.attach(servoPin);
 Serial.begin(115200);


 // Sensor configuration
 while (!Serial)
   ;
 Serial.println("Adafruit BMP388 / BMP390 test");
 bmp.setTemperatureOversampling(BMP3_OVERSAMPLING_8X);
 bmp.setPressureOversampling(BMP3_OVERSAMPLING_8X);
 bmp.setIIRFilterCoeff(BMP3_IIR_FILTER_COEFF_3);
 bmp.setOutputDataRate(BMP3_ODR_50_HZ);
 Serial.println("test 1");
 if (!bmp.begin_I2C()) {
   Serial.println("Could not find a valid BMP3 sensor, check wiring!");
   while (1)
     ;
 }
 if (!bmp.performReading()) {
   Serial.println("Failed to perform reading aw hell naw :(");
   return;
 }
 myServo.write(90);  // Set servo to initial position


 //calibrate altemeter
 for (int i = 0; i < 1000; i++) {
   height_0 = bmp.readAltitude(SEALEVELPRESSURE_HPA);
 }
 digitalWrite(13, HIGH);  // end of setup
}


void loop() {
 height = bmp.readAltitude(SEALEVELPRESSURE_HPA) - height_0;


 if (buttonPressed()) {  // moves it 90 degrees if button was pressed for .12 seconds

     servoPos = 0;
   } else {
     servoPos = 90;

/*
checks if the button is down for .12 seconds
returns true if it is
*/
bool buttonPressed() {
 if (digitalRead(servoButton) != HIGH) {
   delay(25);
   if (digitalRead(servoButton) != HIGH) {
     delay(25);  // this is for accidental presses or accidental contact
     if (digitalRead(servoButton) != HIGH) {
       delay(100);
       if (digitalRead(servoButton) != HIGH) {
         Serial.println("Button pressed");
         delay(20);
         return true;
       }
     }
   }
 }
 return false;
}

/*
returns time since the program has started in seconds
*/
double pTime() {
 return ((double)(millis())) / (double)(1000);
}


/*
records the data
makes important assumption that the change in height per interval doesnt exceed 100!!
*/
void record(float height, double time) {
 String cplusplussucks = "Time from liftoff in seconds is: " + (String)(pTime() - time_0) + " height comparative to liftoff is: " + (String)height + "\n";
 Serial.print(cplusplussucks);
 if (listIndex == 0) {
   EEPROM.write(0, (int)time);
   EEPROM.write(512, (int)height);
 } else {
   EEPROM.write(listIndex, (double)(10 * time));  //records the dime in deciseconds
   EEPROM.write(listIndex + 512, (int)height);    //records the dime in deciseconds
 }
 listIndex++;
}


/*
returns the average of the past sampleLength readings
*/
float getHeight(int sampleLength) {
 float average = 0;
 for (int i = 0; i < sampleLength; i++) {
   average += bmp.readAltitude(SEALEVELPRESSURE_HPA) - height_0;
 }
 if (average / sampleLength <0.0){
   return -average/sampleLength;
 }
 return average / sampleLength;
}




float getDeltaHeight_seperated(float height, float velocity, float dragCoeficciant) {
 float delta_height = 0;
 float dt = 0.25;
 // Simulate the height change while velocity is greater than 1
 while (velocity > 1) {
   float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
   float acceleration = force / mass;
   velocity -= acceleration * dt;
   delta_height += velocity * dt;
 }




 return delta_height;
}




float getDeltaHeight_together(float height, float velocity, float dragCoeficciant, float coastTime) {
 float time = 0;
 float dt = 0.05;
 float delta_height = 0;
 // Simulate the height change during coasting phase
 while (time < coastTime) {
   float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
   float acceleration = force / mass;
   velocity -= acceleration * dt;
   delta_height += velocity * dt;
   time += dt;
 }}
