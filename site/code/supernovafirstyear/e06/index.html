float height_list[5] = {0, 0, 0, 0, 0}; // Keep track of last 5 height values
const float airDensity = 1.297;         // density in kg/m^3
const float referenceArea = 0.038;      // m^2
const float mass = 0.619;               // mass of rocket in kg
const float dragCoefficient = 1.31;     // drag coefficient
int heightDataIndex = 0;


void setup() {
  // Set up serial communication
  Serial.begin(115200);

  // Initialize height_list with the first 5 values of height_data
  for (int i = 0; i < 5; i++) {
    height_list[heightDataIndex] = height_data[heightDataIndex];
    heightDataIndex++;
  }
}


void loop() {
  timetogly();  // Call the function to simulate rocket flight
  float height = 200;
  float time_0 = pTime();
  float velocity = 8.7374;
}


void timetogly() {
  float height = height_data[heightDataIndex];
  float velocity = 0;
  float tApogee = 0;
  float emergencyTime = 100.0;


  // Increment the height data index
  heightDataIndex++;


  // Calculate velocity using the difference of averages of the last 5 height values
  velocity = averageLastFive(height_data, heightDataIndex) - averageLastFive(height_data, heightDataIndex - 5);


  // Calculate predicted apogee
  tApogee = height;
  tApogee += getDeltaHeight_together(height, velocity, .073, 0);
  tApogee += getDeltaHeight_seperated(height, velocity, 1.32);


  // Print the height, velocity, and predicted apogee to the serial monitor
  String cSucks = "height: " + (String)height + ", velocity: " + (String)velocity + ", predicted apogee: " + (String)tApogee;
  Serial.println(cSucks);
}


float getDeltaHeight_seperated(float height, float velocity, float dragCoeficciant) {
  float delta_height = 0;
  float dt = 0.25;

  // Simulate the height change while velocity is greater than 1
  while (velocity > 1) {
    float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
    float acceleration = force / mass;
    velocity -= acceleration * dt;
    delta_height += velocity * dt;
  }


  return delta_height;
}


float getDeltaHeight_together(float height, float velocity, float dragCoeficciant, float coastTime) {
  float time = 0;
  float dt = 0.05;
  float delta_height = 0;

  // Simulate the height change during coasting phase
  while (time < coastTime) {
    float force = .5 * dragCoeficciant * referenceArea * airDensity * velocity * velocity;
    float acceleration = force / mass;
    velocity -= acceleration * dt;
    delta_height += velocity * dt;
    time += dt;
  }


  return delta_height;
}


double pTime() {
  return ((double)(millis())) / (double)(1000); // Get time in seconds
}


float averageLastFive(float data[], int i) {
  // Ensure that the index is large enough to compute the average of the last 5 values
  if (i < 4) {
    Serial.println("Index is too small to compute average of last 5 values.");
    return -1;  // Return -1 or any error value to indicate invalid input
  }


  float sum = 0;

  // Sum the last 5 values starting from index (i-4) to i
  for (int j = i - 4; j <= i; j++) {
    sum += data[j];
  }


  // Return the average of the last 5 values
  return sum / 5.0;
}
